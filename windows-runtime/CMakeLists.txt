cmake_minimum_required(VERSION 3.20)
project(DrMoagi3D VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(DrMoagi3D src/main.cpp)
if(WIN32)
 target_compile_definitions(DrMoagi3D PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(DrMoagi3D PRIVATE d3d11 d3dcompiler dxgi)
endif()
if(MSVC)
 target_compile_options(DrMoagi3D PRIVATE /W4 /EHsc)
else()
 target_compile_options(DrMoagi3D PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME invariants COMMAND DrMoagi3D --self-test)
add_test(NAME cpu_end_to_end COMMAND DrMoagi3D --demo --cpu --tiles 4 --file ${CMAKE_CURRENT_BINARY_DIR}/test_1GB.vmem)
